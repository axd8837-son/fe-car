local function createBlinkerLight(side)
    local pName = (humanoid.RigType == Enum.HumanoidRigType.R15) and (side == "L" and "LeftUpperLeg" or "RightUpperLeg") or (side == "L" and "Left Leg" or "Right Leg")
    local part = character:FindFirstChild(pName) or rootPart
    local att = Instance.new("Attachment", part)
    local l = Instance.new("PointLight", att)
    l.Color, l.Range, l.Brightness, l.Enabled = Color3.fromRGB(255, 130, 0), 10, 6, false
    return l, att
end

local function createBtn(name, pos, size, text, color, key)
    local btn = Instance.new("TextButton", mobileFrame)
    btn.Name, btn.Size, btn.Position = name, size, pos
    btn.BackgroundColor3, btn.BackgroundTransparency = Color3.fromRGB(20, 20, 20), 0.4
    btn.Text, btn.TextColor3, btn.TextScaled = text, Color3.new(1, 1, 1), true
    Instance.new("UICorner", btn)
    local stroke = Instance.new("UIStroke", btn)
    stroke.Thickness, stroke.Color = 2, color
    btn.MouseButton1Down:Connect(function() keysPressed[key] = true end)
    btn.MouseButton1Up:Connect(function()
        keysPressed[key] = false
        if key == "W" and currentSpeed > 50 and isSpecialMode and canPop then
            canPop = false
            for i = 1, 3 do
                local p = Instance.new("Sound", rootPart)
                p.SoundId, p.Volume, p.PlayOnRemove = "rbxassetid://140562715692021", 2, true
                p:Destroy()
                task.wait(math.random(10, 30) / 100)
            end
            task.wait(0.5)
            canPop = true
        end
    end)
    btn.MouseLeave:Connect(function() keysPressed[key] = false end)
    return btn
end

local function createDriftSmoke()
    local emitter = Instance.new("ParticleEmitter")
    emitter.Texture = "rbxasset://textures/particles/smoke_main.dds"
    emitter.Color = ColorSequence.new(Color3.new(1, 1, 1))
    emitter.Transparency = NumberSequence.new({NumberSequenceKeypoint.new(0, 0.2), NumberSequenceKeypoint.new(1, 1)})
    emitter.Size = NumberSequence.new({NumberSequenceKeypoint.new(0, 0.4), NumberSequenceKeypoint.new(1, 1.2)})
    emitter.Lifetime, emitter.Speed, emitter.Rate = NumberRange.new(0.1, 0.2), NumberRange.new(35, 55), 0
    emitter.EmissionDirection, emitter.SpreadAngle = Enum.NormalId.Back, Vector2.new(3, 3)
    emitter.Parent = character:FindFirstChild("LowerTorso") or character:FindFirstChild("Torso") or rootPart
    return emitter
end

userInputService.InputBegan:Connect(function(input, processed)
    if processed then return end
    local k = input.KeyCode
     if k == Enum.KeyCode.V then
        toggleBtn:Activate()
        return
    end
    if k == Enum.KeyCode.W then keysPressed.W = true
    elseif k == Enum.KeyCode.S then keysPressed.S = true
    elseif k == Enum.KeyCode.A then keysPressed.A = true
    elseif k == Enum.KeyCode.D then keysPressed.D = true
    elseif k == Enum.KeyCode.F then keysPressed.F = true
    elseif k == Enum.KeyCode.Left then blinkers.L = not blinkers.L; blinkers.R = false
    elseif k == Enum.KeyCode.Right then blinkers.R = not blinkers.R; blinkers.L = false end
end)

userInputService.InputEnded:Connect(function(input)
    local k = input.KeyCode
    if k == Enum.KeyCode.W then keysPressed.W = false
    elseif k == Enum.KeyCode.S then keysPressed.S = false
    elseif k == Enum.KeyCode.A then keysPressed.A = false
    elseif k == Enum.KeyCode.D then keysPressed.D = false
    elseif k == Enum.KeyCode.F then keysPressed.F = false end
end)

runService.Heartbeat:Connect(function(dt)
    mobileFrame.Visible = not userInputService.KeyboardEnabled
    if not isDriving or not carBase then return end  

    blinkerTimer = blinkerTimer + dt  
    if blinkerTimer >= 0.45 then  
        blinkerTimer, blinkerState = 0, not blinkerState  
        if (blinkers.L or blinkers.R) and blinkerState then blinkSound:Play() end  
    end  

    lightBlinkL.Enabled = blinkers.L and blinkerState  
    strokeBlinkerL.Color = (blinkers.L and blinkerState) and Color3.fromRGB(255, 130, 0) or Color3.fromRGB(200, 200, 200)  
    btnBlinkerL.TextColor3 = (blinkers.L and blinkerState) and Color3.fromRGB(255, 130, 0) or Color3.new(1, 1, 1)  

    lightBlinkR.Enabled = blinkers.R and blinkerState  
    strokeBlinkerR.Color = (blinkers.R and blinkerState) and Color3.fromRGB(255, 130, 0) or Color3.fromRGB(200, 200, 200)  
    btnBlinkerR.TextColor3 = (blinkers.R and blinkerState) and Color3.fromRGB(255, 130, 0) or Color3.new(1, 1, 1)  

    local absSpeed = math.abs(currentSpeed)  
    local isNight = lighting.ClockTime >= 18 or lighting.ClockTime <= 6  
    if lightL and lightR then lightL.Enabled, lightR.Enabled = isNight, isNight end  

    isHandbraking = keysPressed.F    
    local rp = RaycastParams.new()    
    rp.FilterDescendantsInstances, rp.FilterType, rp.RespectCanCollide = {character, carBase}, Enum.RaycastFilterType.Exclude, true   

    if workspace:Raycast(rootPart.Position, rootPart.CFrame.LookVector * 2.5, rp) and absSpeed > 2 then currentSpeed = 0 end    

    local gPush = rootPart.CFrame.LookVector.Y * 1.5  
    if keysPressed.W and not isHandbraking then     
        currentSpeed = math.min(currentSpeed + currentAccel - gPush, currentMaxSpeed)    
    elseif keysPressed.S and not isHandbraking then     
        if currentSpeed > 0 then  
            currentSpeed = math.max(currentSpeed - (currentDecel * math.clamp(1.5 - (absSpeed / currentMaxSpeed), 0.4, 1.2)) - gPush, 0)    
        else  
            currentSpeed = math.max(currentSpeed - currentAccel - gPush, -currentMaxSpeed)  
        end  
    elseif isHandbraking then    
        currentSpeed = (currentSpeed - gPush) * ((absSpeed < 30) and 0.96 or 0.995)     
    else    
        currentSpeed = (currentSpeed - gPush) * (1 - inertiaFriction)     
    end    
  
    local turnStr = isHandbraking and (turningSpeed * handbrakeTurnMultiplier) or turningSpeed    
    local dynTurn = (absSpeed > 0.1) and math.max(0.9, turnStr / (1 + (absSpeed / 100))) or 0    

    if keysPressed.A then targetRotation = targetRotation + dynTurn * dt end    
    if keysPressed.D then targetRotation = targetRotation - dynTurn * dt end    
  
    local isDrifting = isHandbraking and absSpeed > 20 and (keysPressed.A or keysPressed.D)    
    if driftSmokeEmitter then driftSmokeEmitter.Rate = isDrifting and 180 or 0 end    
    if driftSound then driftSound.Volume = isDrifting and 0.8 or 0 end    

    idleSound.Volume = absSpeed < 1 and 1 or 0    
    if absSpeed >= 1 then    
        moveSound.Volume, moveSound.PlaybackSpeed = 1, 1 + (absSpeed / currentMaxSpeed) * 1.5     
    else    
        moveSound.Volume = 0    
    end    
  
    local ground = workspace:Raycast(rootPart.Position, Vector3.new(0, -10, 0), rp)    
    if ground then    
        local comp = targetDistance - (rootPart.Position - ground.Position).Magnitude    
        local vel = (comp - lastVerticalError) / dt    
        lastVerticalError = comp    
        bodyForce.Force = Vector3.new(0, math.max(0, (comp * stiffness) + (vel * damping)), 0)    
        local normal, right = ground.Normal, rootPart.CFrame.RightVector    
        bodyGyro.CFrame = CFrame.fromMatrix(rootPart.Position, right, normal, -right:Cross(normal)) * CFrame.Angles(0, targetRotation - select(2, rootPart.CFrame:ToEulerAnglesYXZ()), 0)    
    else     
        bodyForce.Force, lastVerticalError, bodyGyro.CFrame = Vector3.new(0, 0, 0), 0, CFrame.Angles(0, targetRotation, 0)    
    end    
  
    bodyVelocity.Velocity = bodyVelocity.Velocity:Lerp(rootPart.CFrame.LookVector * currentSpeed, isHandbraking and 0.025 or 0.15)     
    speedText.Text = tostring(math.floor(absSpeed))
end)
